# A small, offline program: records, contracts, tests and explicit errors.
type Request:
    name: str

fn welcome(request: Request) -> str!:
    if len(request.name) == 0:
        fail("name is required")
    return "Hello {request.name}"

fn total(price: money, quantity: int) -> money:
    expect quantity > 0
    return price * quantity

test "invoice":
    assert total(19.90 EUR, 2) == 39.80 EUR

try:
    print(welcome(Request("Ada")))
else e:
    print(e.message)

Expected output

Hello Ada
✓ invoice

rest — Typed local REST routes and validation failures

Local interpreter HTTP tests; not public-server Shield or deployment certification

type Registration:
    name: str
    email: str

server signup:
    port = 8080

    get "/health"():
        return "ready"

    get "/greet/{id}"(id: int, name: str = "visitor"):
        return "hello {name}, number {id}"

    post "/registrations"(entry: Registration):
        return entry

test "routes return text and JSON":
    health = http.get("{signup.url}/health")
    assert health.status == 200
    print(health.text)
    reply = http.post("{signup.url}/registrations", {"name": "Ana", "email": "ana@example.com"})
    assert reply.status == 201
    print(reply.status, reply.json()["name"])

test "path and query parameters are typed":
    reply = http.get("{signup.url}/greet/7?name=Ana")
    print(reply.text)
    try:
        http.get("{signup.url}/greet/not-a-number", retries: 0)
        fail("expected an invalid path parameter")
    else e is HTTPError:
        assert e.message == "GET {signup.url}/greet/not-a-number answered 400 Bad Request"
        print(e.message.replace(signup.url, "URL"))

test "missing record fields fail validation":
    try:
        http.post("{signup.url}/registrations", {"name": "Ana"}, retries: 0)
        fail("expected a missing email field")
    else e is HTTPError:
        assert e.message == "POST {signup.url}/registrations answered 400 Bad Request"
        print(e.message.replace(signup.url, "URL"))

database — In-memory SQLite atomic transfers and rollback

Interpreter in-memory database output regression; no live database certification

## Transfers that either happen whole or not at all.
use nexo.db

db = connect(":memory:")

@table
type Account:
    id: id
    owner: str @unique
    balance: decimal

@table
type Movement:
    id: id
    account: ref Account
    amount: decimal
    note: str

fn transfer(from_owner: str, to_owner: str, amount: decimal) -> bool!:
    expect amount > 0
    db.transaction:
        a = db.Account.find(owner: from_owner)?
        if a.balance < amount:
            fail("{from_owner} has {a.balance}, not {amount}")
        db.Account.update(a, balance: a.balance - amount)
        db.Movement.add(account: a, amount: -amount, note: "to {to_owner}")
        # a failure from here on undoes the lines above too
        b = db.Account.find(owner: to_owner)?
        db.Account.update(b, balance: b.balance + amount)
        db.Movement.add(account: b, amount: amount, note: "from {from_owner}")
    return true

fn balances():
    for a in db.Account.all():
        print("  {a.owner}: {a.balance}")

db.Account.add(owner: "Ana", balance: 100.00)
db.Account.add(owner: "Luis", balance: 20.00)

fn attempt(from_owner: str, to_owner: str, amount: decimal):
    try:
        transfer(from_owner, to_owner, amount)
        print("sent {amount} from {from_owner} to {to_owner}")
    else e:
        print("refused ({e.kind}): {e.message}")

attempt("Ana", "Luis", 30.00)
attempt("Luis", "Ana", 80.00)
attempt("Ana", "Eva", 10.00)

print("only the first one happened:")
balances()
print(db.Movement.count(), "movements")

Expected output

sent 30.00 from Ana to Luis
refused (Error): Luis has 50.00, not 80.00
refused (NotFound): There is no Account with owner = "Eva"
only the first one happened:
  Ana: 70.00
  Luis: 50.00
2 movements

Interpreter process-local test streams; no external Kafka broker tested

## Group the messages of a stream by time.
use nexo.stream

fn per_minute(row):
    print("minute     ", row)

fn last_five(total):
    print("last 5 min ", total)

fn visit(pages):
    print("visit       pages:", pages)

# tumbling: one result per minute and page
kafka("clicks")
  |> window 1m
  |> group by page
  |> count
  |> each per_minute

# sliding: on every message, over the messages of the last 5 minutes
kafka("sales")
  |> sliding 5m
  |> map amount
  |> sum
  |> each last_five

# session: the pages of a visit, which ends after 30 seconds without a click
kafka("visits")
  |> session 30s
  |> count
  |> each visit

test "windows":
    t = time(2026, 10, 28, 9, 0, zone: "UTC")
    clicks = kafka("clicks")
    clicks.send({"page": "/home"}, at: t)
    clicks.send({"page": "/cart"}, at: t + 15s)
    clicks.send({"page": "/home"}, at: t + 40s)
    clicks.send({"page": "/home"}, at: t + 65s)

    sales = kafka("sales")
    sales.send({"amount": 10}, at: t)
    sales.send({"amount": 25}, at: t + 2m)
    sales.send({"amount": 5}, at: t + 6m)

    visits = kafka("visits")
    visits.send({"page": "/home"}, at: t)
    visits.send({"page": "/shoes"}, at: t + 10s)
    visits.send({"page": "/pay"}, at: t + 20s)
    visits.send({"page": "/home"}, at: t + 5m)

Expected output

Fake agent tests and direct tool calls; no live model, email or provider evaluated

## An agent that answers order questions with the program's own functions.
type Order:
    id: int
    status: "paid" | "shipped" | "delivered"
    email: str

orders = [
    Order(id: 7, status: "shipped", email: "ana@mail.com"),
    Order(id: 8, status: "paid", email: "luis@mail.com")]

@tool "Looks up an order by its number"
fn find_order(id: int) -> Order!:
    for o in orders:
        if o.id == id:
            return o
    fail("there is no order {id}")

@tool "Sends a short email to the customer of an order"
fn email_customer(id: int, text: str) -> str!:
    o = find_order(id)?
    return "sent to {o.email}"

agent support:
    model = "anthropic:claude-opus"
    tools = [find_order, email_customer]
    goal = "Answer questions about orders, politely and briefly"
    max_steps = 5
    confirm = [email_customer]

test "the tools are ordinary functions":
    print(find_order(7).status)
    try:
        find_order(99)
    else e:
        print(e.message)

test "the agent's answer can be faked":
    fake support.handle returns "Order 7 has shipped."
    print(support.handle("Where is my order 7?"))

Expected output

shipped
there is no order 99
✓ the tools are ordinary functions
Order 7 has shipped.
✓ the agent's answer can be faked

Know the boundaries

Use this as assistance, not a certification. A passing check does not prove runtime permissions, kernel isolation, production readiness or correct model behavior. Roadmap items are not available APIs.

Implemented — core

Python interpreter and Rust/LLVM native compiler; records, explicit errors, contracts and tools. Package versions differ; do not infer a stable Nexo 1.0 release.

Implemented — diagnostics

Versioned check --json diagnostics and conservative --effects analysis. Unknown/dynamic effects remain unknown; static checks do not grant permissions.

Implemented — agents

Typed input/output, registered tools, approval before conversion, inherited cooperative cancellation/deadlines and reported token/money reservations. Counters trust configured prices/usage; they are not invoice guarantees or arbitrary-client metering.

Implemented — recovery

Explicit version 2 operator grants enable encrypted bounded agent recovery and cross-process accounting. Native persistence is Linux AMD64 only. Durable inputs are text/none; result defaults and durable MCP calls are rejected. Uncertain dispatch fails closed; no exactly-once external-effect guarantee.

Implemented — shield

Policy/API enforcement, dependency/dispatch ceilings and optional hardened containers/Linux workers exist. Kernel isolation requires supported host facilities; API-only tests prove no sandbox-escape guarantee. Generic workflow persistence remains rejected under Shield.

Implemented — telemetry

NEXO_AGENT_TRACE=stderr explicitly enables payload-free OTLP JSON agent spans. No exporter runs by default. It is not universal automatic telemetry or a monitoring service.

Implemented — libraries

The API catalog and working examples cover web/HTTP, SQL/document databases, streams, crypto, privacy, infrastructure and CI. Inspect exact signatures and connector/profile restrictions; an integration or emulator test is not a production certification.

Implemented — mcp

Developer MCP server and agent MCP client exist. Published-spec interoperability and lifecycle/authorization completeness remain unverified; the roadmap audit is not completed by their presence.

Experimental — web-target

Browser playground and a platform-limited WebAssembly/web target exist. Do not describe them as production standalone WASI deployment or unrestricted native-library parity.

Planned — interop-roadmap

AI-native Phase 4 interoperability work (including a justified A2A adapter and standalone WASI investigation) and Phase 5 Nexo Sentinel validation remain future work. ml/ui/notify are not available modules.

Planned — registry

Signed packages/static registries exist, but an official registry service and reviewed-package ecosystem are not established.

Unverified — operations

Live cloud readiness, HA/SLO achievement, local Kubernetes compatibility, independent security audits, current provider prices and model-specific token parity require separate evidence. Existing infra adapters/examples do not prove these outcomes.

Shield Security Assessment

Every architecture and readiness review assesses Shield as the preferred security foundation. Start with least privilege; verify workload and host compatibility before choosing API enforcement, Linux offline/online workers or hardened Docker. Required isolation must never silently fall back.

Open the versioned security checklist and profile limits. Report the recommendation, capabilities, execution profile, unsupported operations, isolation limits, validation commands and remaining risks. Load this detail only when relevant; offline assistants can ask you to supply its text. Shield complements Kubernetes and infrastructure controls.

How versions and measurements work

The content revision hashes the guide rules, generator, page assets and selected repository evidence. Identical inputs produce identical prompts. Existing snapshots are retained when context changes. Package versions describe different components, not a stable Nexo 1.0 release.

The llms.txt format is a documentation discovery proposal; automatic adoption by an LLM is not guaranteed. Token counts use the documented tiktoken tokenizer, not a universal token estimate. See the maintenance guide.
